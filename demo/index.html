Send yourself a notification.

One message, one on-chain event, one browser notification. Set up this browser, then send yourself a message through the echo realm, a small gno.land contract that emits each message as an event addressed to its sender.

  1. Set up this browser

    Turn on notifications, live updates, or both. Either one is enough for step 2.

    Live updates

    This page shows your message the moment it arrives, the way a dapp updates its own view. No permission prompt, and it works in any browser. While the page is open, it keeps a connection to the gnotif server and sends it your address.

    import { Gnotif } from "gnotif";

const gnotif = new Gnotif({ network: "onyx", icon: "/icon.png" });
const [echo] = await gnotif.triggers("");

button.addEventListener("click", async () => {
  await gnotif.enable();
  await gnotif.setOptins([{ trigger: echo.id, value: address }]);
});

gnotif.on("notification", (n) => show(n));
live.addEventListener("change", async () => {
  if (!live.checked) return gnotif.disconnect();
  await gnotif.setOptins([{ trigger: echo.id, value: address }]);
  await gnotif.connect();
});

      func init(cur realm) {
	me := cur.Sub(gnotif.Subpath)
	h := gnotif.Open(cross(me))
	h.Set(cross(me), "echo", gnotif.Trigger{Event: "Echo", Param: "to", Title: "Echo", Body: "{msg}", Link: "/"})
}

func Echo(cur realm, msg string) {
	chain.Emit("Echo", "to", cur.Previous().Address().String(), "msg", msg)
}
